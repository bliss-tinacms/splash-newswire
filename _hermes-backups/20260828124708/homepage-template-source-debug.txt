15|function slugifyFilename(value?: string | null): string {
16|  if (!value || typeof value !== "string") return "untitled";
17|  let input = value.trim().toLowerCase();
18|  if (!input) return "untitled";
19|
20|  if (input.startsWith("http://") || input.startsWith("https://")) {
21|    const parts = input.split("/");
22|    input = parts.slice(3).join("/");
23|  }
24|
25|  input = input.split("?")[0].split("#")[0];
26|  input = input.split("\\").join("/");
27|  while (input.startsWith("/")) input = input.slice(1);
28|  while (input.endsWith("/")) input = input.slice(0, -1);
29|  if (input.startsWith("blog/")) input = input.slice(5);
30|
31|  let output = "";
32|  let lastWasDash = false;
33|  for (const ch of input) {
34|    const isLetter = ch >= "a" && ch <= "z";
35|    const isNumber = ch >= "0" && ch <= "9";
36|    const isSlash = ch === "/";
37|    const isAllowed = isLetter || isNumber || isSlash || ch === "-" || ch === "_" || ch === ".";
38|    if (isAllowed) {
39|      output += ch;
40|      lastWasDash = false;
41|    } else if (!lastWasDash) {
42|      output += "-";
43|      lastWasDash = true;
44|    }
45|  }
46|
47|  while (output.startsWith("-")) output = output.slice(1);
48|  while (output.endsWith("-")) output = output.slice(0, -1);
49|  while (output.includes("//")) output = output.split("//").join("/");
50|  return output || "untitled";

53|function toPublicSlug(value?: string | null): string {
54|  if (!value || typeof value !== "string") return "";
55|  let input = value.trim().toLowerCase();
56|  if (!input) return "";
57|
58|  if (input.startsWith("http://") || input.startsWith("https://")) {
59|    const parts = input.split("/");
60|    input = parts.slice(3).join("/");
61|  }
62|  input = input.split("?")[0].split("#")[0];
63|  input = input.split("\\").join("/");
64|  while (input.startsWith("/")) input = input.slice(1);
65|  while (input.endsWith("/")) input = input.slice(0, -1);
66|  if (input.startsWith("blog/")) input = input.slice(5);
67|
68|  let out = "";
69|  let dash = false;
70|  for (const ch of input) {
71|    const ok = (ch >= "a" && ch <= "z") || (ch >= "0" && ch <= "9");
72|    if (ok) { out += ch; dash = false; }
73|    else if (!dash) { out += "-"; dash = true; }
74|  }
75|  while (out.startsWith("-")) out = out.slice(1);
76|  while (out.endsWith("-")) out = out.slice(0, -1);
77|  return out;
